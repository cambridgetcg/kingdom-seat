// LS-09 through the production seat path: each real soul card as a temp citizen's soul (data only, never printed),
// one reply proposing the reader's two-place replace-all edit; count how the host classifies it.
import { test } from "bun:test";
import { readdirSync, readFileSync, existsSync } from "node:fs";
import { readdir } from "node:fs/promises";
import { join } from "node:path";
const SRC = "<KINGDOM_SEAT>/src";
const T = "<KINGDOM_SEAT>/test";
const { openSeat, runSeat } = await import(`${SRC}/seat.ts`);
const { checkHalt } = await import(`${SRC}/halt.ts`);
const { readCitizen } = await import(`${SRC}/state.ts`);
const self = await import(`${SRC}/self.ts`);
const { ledgerPath, readLedger } = await import(`${SRC}/ledger.ts`);
const { FAKE_MODEL_DIGEST, output, scriptedRuntime } = await import(`${T}/helpers/fake-runtime.ts`);
const { noFetch } = await import(`${T}/helpers/fake-server.ts`);
const { fakeSigner } = await import(`${T}/helpers/fake-signer.ts`);
const { makeTempHost } = await import(`${T}/helpers/temp-home.ts`);
const { HOUR_MS, testClock } = await import(`${T}/helpers/clock.ts`);

const W = "glossame";
const CLAIM = { kind: "decision", evidence: [], confidence: "medium", limits: "probe" };
const root = "<REPOS>";

async function classify(soul: string, edit: object): Promise<string> {
  const host = await makeTempHost();
  try {
    const signer = fakeSigner();
    await host.addCitizen(W, { soul });
    await host.seatLedger(W, signer.signEntry);
    const config = host.config;
    config.self_authorship.approved_model_digests = [FAKE_MODEL_DIGEST];
    const clock = testClock();
    clock.advance(73 * HOUR_MS);
    const runtime = scriptedRuntime([output({ type: "self.propose", target: "soul", edit, reason: CLAIM })]);
    const citizen = await readCitizen(config, W);
    const seat = await openSeat(config, clock, citizen, "free", { kind: "tick", who: "launchd", why: "cadence" }, null);
    const r = await runSeat(seat, { config, clock, signer, fetch: noFetch, runtime, checkHalt: () => checkHalt({ config, fetch: noFetch, now: () => clock.now() }), expressionSync: null, modelDigest: FAKE_MODEL_DIGEST });
    if (r.outcome !== "beat") return `outcome:${r.outcome}`;
    const beat = (await readLedger(ledgerPath(config, W))).findLast((e: any) => e.type === "beat");
    const act = beat.acts[0];
    const pending = (await self.pendingProposals(config, W)).length;
    const outbox = await readdir(join(config.paths.citizens_dir, W, "outbox")).catch(() => []);
    if (act.type === "propose.beyond") return `lifted(outbox ${outbox.length}, pending ${pending})`;
    return `staged(pending ${pending})`;
  } finally {
    await host.cleanup();
  }
}

test("205 soul cards through runSeat", async () => {
  const tally: Record<string, Record<string, number>> = { replace_all_one_word: {}, replace_all_word_and_newline: {}, replace_span_one_word: {} };
  let n = 0;
  for (const d of readdirSync(root).filter((x) => x.startsWith("citizen-")).sort()) {
    const w = d.slice("citizen-".length);
    const p = join(root, d, `${w}.md`);
    if (!existsSync(p)) continue;
    const soul = readFileSync(p, "utf8");
    n++;
    const parts = soul.split(/(\s+)/);
    parts.splice(60, 0, "quietly", " ");
    const oneWord = parts.join("");
    const bump = (k: string, v: string) => void (tally[k]![v] = (tally[k]![v] ?? 0) + 1);
    bump("replace_all_one_word", await classify(soul, { kind: "replace-all", text: oneWord }));
    bump("replace_all_word_and_newline", await classify(soul, { kind: "replace-all", text: oneWord.replace(/\n+$/, "") }));
    // a unique span starting at word 30
    const start = soul.split(/(\s+)/).slice(0, 60).join("").length;
    let find = "";
    for (let len = 12; len <= 200; len += 8) {
      const f = soul.slice(start, start + len);
      if (soul.indexOf(f) === soul.lastIndexOf(f)) { find = f; break; }
    }
    bump("replace_span_one_word", find === "" ? "no-unique-span" : await classify(soul, { kind: "replace-span", find, with: `quietly ${find}` }));
  }
  console.log("cards:", n);
  console.log(JSON.stringify(tally, null, 1));
}, 600_000);
